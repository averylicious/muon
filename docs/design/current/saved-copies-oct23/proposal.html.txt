}
function render() {
  closeMenu(); hideSnack();
  let h = '';
  if (state.card) h += `<div class="banner" role="status">${svg(I.sd)}<div><b>SD card not found</b>Copies on it are hidden and stay on the card. Put the same card back, then try again.<button id="cardretry">Try again</button></div></div>`;
  if (!all().length) {
    h += `<div class="empty"><div class="shape" aria-hidden="true"></div><h3>Nothing on this device yet</h3>
      <p>Long-press a song and choose Download to keep a copy here. Songs you listen to may also be kept for a while.</p></div>`;
    scroll.innerHTML = h; return;
  }
  h += `<button class="chiprow" id="why">${svg(I.info)}<span>Plays from this device</span><b>Details${svg(I.chevron)}</b></button>`;
  h += `<div class="section-head"><span class="label">Downloads</span><span class="count">· ${state.downloads.length}</span></div>`;
  h += state.downloads.map(row).join('');
  if (state.listening.length) {
    h += `<button class="section-head" id="toggle" aria-expanded="${state.listeningOpen}">
      <span class="label">Kept from listening</span><span class="count">· ${state.listening.length}</span><span class="spacer"></span>
      <span class="toggle">${state.listeningOpen ? 'Hide' : 'Show'}${svg(state.listeningOpen ? I.collapse : I.expand)}</span></button>`;
    h += `<p class="section-sub">Saved automatically while you listened. When the listening cache is full, older ones make room.</p>`;
    if (state.listeningOpen) h += state.listening.map(row).join('');
  }
  h += `<div class="pager">${all().length} on this device</div>`;
  scroll.innerHTML = h;
}

// Row menu: no disabled items. Delete is always offered for a listed copy.
let menuEl = null, menuFor = null;
function closeMenu() { if (menuEl) { menuEl.remove(); menuEl = null; menuFor = null; } }
function openMenu(id, btn) {
  closeMenu();
  const s = all().find(x => x.id === id);
  const m = document.createElement('div');
  m.className = 'menu'; m.setAttribute('role', 'menu'); m.setAttribute('aria-label', `Options for ${s.title}`);
  let h = `<button role="menuitem" data-act="play">${svg(I.play)}Play</button>
    <button role="menuitem" data-act="details">${svg(I.details)}Details</button>`;
  if (s.update === 'eligible') h += `<button role="menuitem" data-act="update">${svg(I.swap)}Update storage format</button>`;
  h += `<hr><button role="menuitem" data-act="delete">${svg(I.del)}${s.failed ? 'Try delete again' : s.source === 'played' ? 'Delete from device' : 'Delete download'}</button>`;
  m.innerHTML = h;
  const phone = document.getElementById('phone');
  phone.appendChild(m);
  const r = btn.getBoundingClientRect(), p = phone.getBoundingClientRect();
  let top = r.bottom - p.top - 6;
  if (top + m.offsetHeight > p.height - 150) top = r.top - p.top - m.offsetHeight + 6;
  m.style.top = top + 'px';
  menuEl = m; menuFor = id;
  m.querySelector('button').focus();
  m.addEventListener('click', e => {
    const act = e.target.closest('button')?.dataset.act; if (!act) return;
    closeMenu();
    if (act === 'play') note.textContent = `Preview: “${s.title}” would play from this device.`;
    if (act === 'details') details(s);
    if (act === 'update') updateDialog(s);
    if (act === 'delete') deleteDialog(s);
  });
}

const scrim = document.getElementById('scrim'), sheet = document.getElementById('sheet'), dialog = document.getElementById('dialog');
const snack = document.getElementById('snack');
function hideAll() { scrim.classList.remove('show'); sheet.classList.remove('show'); dialog.classList.remove('show'); }
function hideSnack() { snack.classList.remove('show'); }
scrim.onclick = hideAll;
function details(s) {
  const kept = s.source === 'played'
    ? `${svg(I.history)}<span>Kept from listening<small>Saved automatically. It can make room for newer songs.</small></span>`
    : `${svg(I.done)}<span>Download<small>You chose to keep this.</small></span>`;
  const where = s.shelf === 'card' ? `${svg(I.sd)}<span>SD card · ${s.size}</span>` : `${svg(I.phone)}<span>Phone · ${s.size}</span>`;
  const fmt = s.source === 'played' ? '' :
    `<dt>STORAGE FORMAT</dt><dd>${svg(I.swap)}<span>${s.update === 'eligible' ? 'Older format · update available<small>Needs temporary free space for a second copy while it’s checked.</small>' : 'Current'}</span></dd>`;
  const last = s.failed ? `<dt>LAST DELETE</dt><dd>${svg(I.warn)}<span>Didn’t finish<small>The SD card was removed. Put it back and try again.</small></span></dd>` : '';
  sheet.innerHTML = `<div class="handle"></div>
    <div class="head"><div class="art ${s.art}" aria-hidden="true"></div><div><div class="title">${s.title}</div><div class="artist">${s.artist}</div></div></div>
    <dl><dt>KEPT AS</dt><dd>${kept}</dd><dt>STORED ON</dt><dd>${where}</dd>
    <dt>PLAYS FROM</dt><dd>${svg(I.info)}<span>This device only<small>Muon can’t confirm it still matches the song in Tauon.</small></span></dd>
    ${fmt}${last}</dl>
    <div class="actions"><button class="btn filled" data-a="play">${svg(I.play)}Play</button>
    ${s.update === 'eligible' ? `<button class="btn tonal" data-a="update">${svg(I.swap)}Update format</button>` : ''}
    <button class="btn outlined" data-a="delete">${svg(I.del)}${s.failed ? 'Try again' : 'Delete'}</button></div>`;
  scrim.classList.add('show'); sheet.classList.add('show');
  sheet.onclick = e => { const a = e.target.closest('button')?.dataset.a; if (!a) return;
    if (a === 'play') { hideAll(); note.textContent = `Preview: “${s.title}” would play from this device.`; }
    if (a === 'update') updateDialog(s); if (a === 'delete') deleteDialog(s); };
}
function why() {
  sheet.innerHTML = `<div class="handle"></div><h3>Why these play from here</h3><div class="prose">
    <p>Tauon can give a song’s number to a different song after its library changes. Muon can’t confirm that a copy here still matches what Tauon has now.</p>
    <p>So these play only from this device, and your Tauon library never uses them in its place. You can still play or delete any of them.</p></div>
    <div class="actions"><button class="btn tonal">Got it</button></div>`;
  scrim.classList.add('show'); sheet.classList.add('show');
  sheet.onclick = e => { if (e.target.closest('button')) hideAll(); };
}
function showDialog(title, body, confirm, onConfirm) {
  dialog.innerHTML = `<h3>${title}</h3><p>${body}</p><div class="buttons"><button data-b="cancel">Cancel</button><button data-b="ok">${confirm}</button></div>`;
  sheet.classList.remove('show'); scrim.classList.add('show'); dialog.classList.add('show');
  dialog.querySelector('[data-b=ok]').focus();
  dialog.onclick = e => { const b = e.target.closest('button')?.dataset.b; if (!b) return; hideAll(); if (b === 'ok') onConfirm(); };
}
function deleteDialog(s) {
  const where = s.shelf === 'card' ? 'SD card' : 'phone';
  showDialog(s.source === 'played' ? 'Delete from this device?' : 'Delete this download?',
    `“${s.title}” will be deleted from this ${where}. Other copies here and your Tauon library aren’t changed.`,
    'Delete', () => { note.textContent = 'Preview only: nothing was deleted. In the app the row would disappear once the delete finishes.'; });
}
function updateDialog(s) {
  showDialog('Update storage format?',
    `Muon makes a new copy of “${s.title}” in the current format and checks it before replacing this one. It needs temporary space for a second copy. You can follow or cancel it in Settings.`,
    'Update', () => { note.textContent = 'Preview only: no storage was changed.'; });
}
